function buildStrengthSessions(primary, correction, effortLimit, strainBand){ const cap = effortLimit; // Decide block style from goal // primary is LOWER/HINGE/PULL/PUSH/BALANCED (goal-driven) const isHypertrophyLower = (document.getElementById("goal").value === "lower"); // Volume scaling based on strain const volumeScale = (strainBand === "HIGH") ? 0.65 : (strainBand === "MEDIUM") ? 0.80 : 1.00; // Helper: rounds to clean set counts const scaledSets = (base) => Math.max(2, Math.round(base * volumeScale)); // Pick focus order const primaryKey = (primary === "BALANCED") ? "LOWER" : primary; const secondaryKey = correction ? correction : pickSecondary(primaryKey); const tertiaryKey = pickSecondary(secondaryKey); // --- HYPERTROPHY (glutes/legs) template --- // This is goal-driven: when goal is lower-body growth, sessions switch to growth ranges and more lower-body work. function hypertrophyLowerSession(title, emphasis){ const topSetEffort = Math.min(9, Math.round(cap)); // still safe under fatigue const pumpEffort = Math.max(7, Math.round(cap-2)); // base weekly structure: 2–3 lower sessions, each with ~8–14 "work sets" total (scaled by sport load) // NOTE: you can later swap these to exercise pickers (dropdowns) per slot. const A = scaledSets(emphasis === "glute" ? 4 : 3); const B = scaledSets(3); const C = scaledSets(3); return { title, lines: [ "Main (glutes/legs): " + A + "×6–10 at effort " + topSetEffort + "/10", "Support (hinge or squat): " + B + "×8–12 at effort " + pumpEffort + "/10", "Accessory (glute pump): " + C + "×10–15 at effort " + pumpEffort + "/10", "If you climb/fight hard this week: keep 1–2 reps in reserve on accessories." ] }; } // --- STRENGTH template (fallback for non-hypertrophy goals) --- function strengthSession(focus){ const mainHard = (strainBand === "HIGH") ? "3×4–6" : "4×4–6"; const second = "3×6–8"; const easy = "2–3 easy sets (optional)"; const reduceLine = (strainBand !== "LOW") ? "Reduce extra exercises this week." : "Add a little load if it feels good."; return { title: (isProMode ? focus : prettyPillar(focus)), lines: [ "Main work (" + focus + "): " + mainHard + " at effort " + Math.min(9, Math.round(cap)) + "/10", "Support work: " + second + " at effort " + Math.max(7, Math.round(cap-1)) + "/10", "Small extras: " + easy, reduceLine ] }; } // Build sessions let sessions = []; if(isHypertrophyLower){ // Make the plan obviously “glute growth” // Session 1: glute emphasis sessions.push(hypertrophyLowerSession("Lower (glutes) — growth", "glute")); // Session 2: minimum correction if needed, otherwise upper push/pull maintenance if(correction){ sessions.push(strengthSession(correction)); } else { sessions.push({ title: "Upper body — keep strong", lines: [ "Choose PUSH or PULL: " + scaledSets(3) + "×6–10 at effort " + Math.max(7, Math.round(cap-1)) + "/10", "Second move: " + scaledSets(3) + "×8–12 at effort " + Math.max(7, Math.round(cap-2)) + "/10", "Keep it short. Save energy for sport + glutes." ] }); } // Session 3: legs/hinge growth (if strain allows) if(strainBand === "LOW"){ sessions.push(hypertrophyLowerSession("Lower (legs/hinge) — growth", "leg")); } else { // If strain is medium/high: don’t force the 3rd long session sessions.push({ title: "Optional mini glute session (10–20 min)", lines: [ "1–2 glute exercises: " + scaledSets(2) + "×10–15", "Stop before you feel destroyed.", "This is to keep frequency, not to crush you." ] }); } return sessions; } // Non-hypertrophy goals = strength base sessions.push(strengthSession(primaryKey)); sessions.push(strengthSession(secondaryKey)); if(strainBand === "LOW"){ sessions.push(strengthSession(tertiaryKey)); } return sessions; // Helpers function prettyPillar(k){ return ({ LOWER:"Lower (legs/glutes)", HINGE:"Hinge (hips)", PULL:"Pull (back/grip)", PUSH:"Push (pressing)" }[k] || k); } function pickSecondary(current){ const options = ["LOWER","HINGE","PULL","PUSH"].filter(x => x !== current); return options[0]; } }
